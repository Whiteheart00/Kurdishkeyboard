[zikr]
وَ ‌عَلَيْكُمَ ‌‌السَّلَاْمُ | وَ ‌عَلَيْكُمَ ‌‌السَّلَاْمُ‌ ‌وَ ‌‌رَحْمَةُ ‌‌اللّٰهِ ‌‌وَ ‌‌بَرَكَاتُهُ;;
السَّلَاْمُ ‌‌عَلَیْکُم‌ | السَّلَاْمُ ‌‌عَلَیْکُم‌ ‌وَ ‌‌رَحْمَةُ‌‌ اللّٰهِ‌ وَ ‌‌بَرَکَاْتُه‌;;
بِسْمِ‌ اللَّهِ ‌الرَّحْمَنِ ‌الرَّحِيمِ;;
أ‌َشهَدُ ‌أَن‌ ‌لا‌َ ‌إِلٰهَ إِلاَّ ‌الله| أ‌َشهَدُ ‌أَن‌ ‌لا‌َ ‌إِلٰهَ ‌‌إِلاَّ ‌الله ‌‌و ‌‌أَشْهَدُ ‌‌أَنَّ ‌‌ﷴ ‌‌عَبْدُهُ ‌‌و ‌‌رَسُولُە;;
لا إلهَ ‌إلاَّ اللَّە وحْدهُ | لا إلهَ‌ ‌إلاَّ ‌ ‌اللَّه‌ وحْدهُ‌ ‌‌لاَ ‌‌شَرِيكَ‌‌ ‌لهُ‌‌ ‌لَهُ‌‌ ‌المُلْكُ‌‌ ‌ولَهُ‌ ‌‌الحمْدُ‌ ‌‌وَ‌ هُو‌‌ ‌عَلَى‌‌ كُل‌ِّ ‌‌شَيءٍ ‌‌قَدِيرٌ;;
اَللَّه‌‌ ‌أَكْبَرُ ‌‌كَبِيرا‌ً | اَللَّه‌‌ ‌أَكْبرُ ‌‌كَبِيرا‌ً‌‌ ‌وَ‌الْحَمْد‌ُ ‌‌لِلَّهِ ‌‌كَثِيراً ‌‌وسُبْحَانَ‌ ‌‌اَللَّهِ‌ ‌‌بِكُرَةً ‌‌وَأَصِيلاً;;
حَسْبِيَ ‌‌اللّٰهُ ‌‌وَنِعْمَ ‌‌الْوَكِيل;;
حَسْبِيَ‌‌ اللَّهُ ‌‌لَا ‌‌إِلَٰهَ ‌‌إِلَّا ‌‌هُوَ ‌‌عَلَيْهِ ‌‌تَوَكَّلْتُ ‌‌وَهُوَ ‌‌رَبُّ ‌‌الْعَرْشِ ‌‌الْعَظِيمِ;;
سُبْحَانَكَ‌ ‌‌اللَّهُمَّ‌ | سُبْحَانَكَ‌ ‌‌اللَّهُمَّ‌ ‌‌وَبِحَمْدِكَ‌ ‌‌أَشْهَدُ ‌‌أَنَ ‌‌لا‌‌ إِلهَ‌ ‌‌إِلَّا‌ ‌أَنْتَ ‌‌أَسْتَغْفِرُكَ ‌‌وَأَتْوبُ ‌‌إِلَيْكَ;;
لَّا ‌‌إِلَٰهَ‌‌ ‌إِلَّا‌ ‌أَنتَ‌‌ ‌سُبْحَانَكَ‌‌ إِنِّي‌‌ كُنتُ‌‌ ‌مِنَ‌‌ ‌الظَّالِمِينَ;;
إِنَّا‌‌ لِلّهِ‌‌ وَ إِنَّـا‌‌ إِلَيْهِ‌‌ ‌رَاجِعونَ;;
لَا‌‌ حَوْلَ‌‌ وَ ‌‌لَا ‌قُوَّةَ‌‌ ‌إلَّا‌‌ ‌بِاَللَّهِ‌‌ الْعَلِيِّ‌‌ الْعظِيمِ;;
اللَّهُ ‌‌لَا ‌‌إِلَٰهَ ‌‌إِلَّا ‌‌هُوَ ‌‌الْحَيُّ ‌‌الْقَيُّومُ ‌‌لَا ‌‌تَأْخُذُهُ ‌‌سِنَةٌ ‌‌وَلَا ‌‌نَوْمٌ ‌‌لَهُ ‌‌مَا ‌‌فِي ‌‌السَّمَاوَاتِ ‌‌وَمَا ‌‌فِي ‌‌الْأَرْضِ ‌‌مَنْ ‌‌ذَا ‌‌الَّذِي ‌‌يَشْفَعُ ‌‌عِنْدَهُ ‌‌إِلَّا ‌‌بِإِذْنِهِ ‌‌يَعْلَمُ ‌‌مَا ‌‌بَيْنَ ‌‌أَيْدِيهِمْ ‌‌وَمَا ‌‌خَلْفَهُمْ ‌‌وَلَا ‌‌يُحِيطُونَ ‌‌بِشَيْءٍ ‌‌مِنْ ‌‌عِلْمِهِ ‌‌إ‌ِلَّا ‌‌بِمَا ‌‌شَاءَ ‌‌وَسِعَ ‌‌كُرْسِيُّهُ ‌‌السَّمَاوَاتِ ‌‌وَالْأَرْضَ ‌‌وَلَا ‌‌يَئُودُهُ ‌‌حِفْظُهُمَا ‌‌وَهُوَ ‌‌الْعَلِيُّ ‌‌الْعَظِيمُ\u06DD٢٥٥;;
قَالَ‌‌ ‌إِنَّمَآ‌‌ ‌أَشْكُواْ‌‌ بَثِّى‌‌ وَحُزْنِىٓ‌‌ إِلَى‌‌ اللَّهَُ;;
أَعُوذُ‌ ‌بكَلِمَاتِ ‌‌اللَّهِ ‌التَّامَّةِ، ‌مِن‌ كُلِّ ‌شيطَانٍ ‌وهَامَّةٍ، ‌و ‌مِنْ‌ كُلِّ ‌عَيْنٍ ‌‌لَامَّةٍ;;
أَعُوذُ‌‌ بِاللَّه‌‌ ‌مِنَ‌‌ ‌الشَّيطَانِ‌‌ ‌الرَّجِيمِ;;
اللَّهُمَّ ‌‌لَا ‌‌سَهْلَ ‌‌إِلَّا ‌‌مَا ‌‌جَعَلْتَهُ ‌‌سَهْلًا، وَأَنْتَ‌‌ تَجْعَلُ ‌‌الْحَزْنَ ‌‌إِذَا ‌‌شِئْتَ ‌‌سَهْلًا;;
اللَّهُمَّ‌ ‌‌أَعِزَّ‌ ‌‌الْإِسْلَامَ‌ ‌‌وَالْمُسْلِمِينَ‌ ‌‌وَأَذَل‌ ‌‌الشِّرْكَ‌ ‌‌وَالْمُشْرِكِين‌ ‌‌وَدمِر‌ ‌‌أَعْدَاءَ‌ ‌‌الدِّينِ‌ ‌‌وَانْصُر‌ ‌‌عِبَادِكَ‌‌ ‌الْمُوَحِّدِين;;
اللَّهُمَّ‌‌ ‌اکفِني‌ ‌‌بحلالِكَ‌ ‌‌عَن‌ ‌‌حرامِك‌ ‌‌وأغنِني‌ ‌‌بفضلِك‌ ‌‌عمَّن‌ ‌‌سواكَ;;
اَللّٰهُمَّ ‌‌أَعِنِّيْ ‌‌عَلَىٰ ‌‌ذِكْرِكَ ‌‌وَشُكْرِكَ ‌‌وَحُسْنِ‌‌ عِبَادَتِكَ;;
اللَّهُمَّ‌‌ ‌أصْلِحْ ‌‌لِي‌ ‌‌دِينِي‌ ‌‌الَّذِي‌ ‌‌هُوَ‌ ‌‌عِصْمَةُ‌ ‌أَمْرِي‌، ‌‌وَأصْلِحْ‌ ‌‌لِي‌ ‌‌دُنْيَايَ‌ ‌‌الَّتِي‌ ‌‌فِيهَا‌ ‌‌مَعَاشِي‌‌ ‌وَأصْلِحْ‌ ‌‌لِي‌ ‌آخِرَتي‌‌ ‌الَّتِي‌ ‌‌فيهَا‌ ‌‌مَعَادِي‌، ‌‌وَاجْعَلِ‌‌ ‌الحَيَاةَ‌ ‌‌زِيَادَةً‌ ‌‌لِي‌ ‌‌فِي‌‌ كُلِّ ‌‌خَيرٍ‌‌ ‌وَاجْعَلِ‌ ا‌لمَوتَ‌ ‌‌رَاحَةً‌ ‌‌لِي‌‌ ‌مِنْ‌‌ كُلِّ‌‌ ‌شَرٍّ;;
اللَّهُمَّ‌‌ رَبَّ ‌‌النَّاسِ ‌‌أذْهِبِ ‌‌البَاسَ، اشْفِهِ ‌‌وأَنْتَ ‌‌الشَّافِي، ‌لا ‌‌شِفَاءَ‌ ‌‌إلَّا‌ شِفَاؤُكَ، ‌شِفَاءً لا يُغَادِرُ‌ سَقَماً;;
اللَّهُمَّ‌ إنِّي‌ أَعُوذُ ‌بِكَ مِنَ العَجْزِ وَالكَسَلِ، وَالجُبْنِ وَالهرَمِ وَالبُخْلِ، وَ أَعُوذُ بِكَ مِنْ عَذَابِ القَبْرِ، وَ مِنْ فِتْنَةِ المَحْيَا وَالمَمَاتِ;;
اللَّهُمَّ إنِّي أَعُوذُ بِكَ مِنَ الهَمِّ وَالحَزَنِ، وَالعَجْزِ وَالكَسَلِ، وَالجُبْنِ وَالبُخْلِ، وَضَلَعِ الدَّيْنِ، وَغَلَبَةِ الرِّجَالِ;;
اللَّهُمَّ‌ إنِّي أَعُوذُ بِكَ مِنَ البَرَصِ وَالجُنُونِ وَالجُذَامِ وَمِنْ سَيِّئِ الأَسْقَامِ;;
اللَّهُمَّ ‌‌أَحْسِنْ ‌‌عَاقِبَتِنَا ‌‌فِي ‌‌الْأُمُورِ ‌‌كُلِّهَا، وَأَجِرْنَا‌‌ مِنْ ‌‌خِزْيِ ‌‌الدُّنْيَا ‌‌وَعَذَابِ ‌‌الْآخِرَةِ;;
اللَّهمَّ ‌‌اغفِر ‌‌لي ‌‌ذنبي ‌‌كُلَّهُ، دِقَّهُ‌‌ وجِلَّهُ، وأوَّلَهُ‌‌ وآخرَهُ، وعلانيتَهُ‌‌ وسِرَّهُ;;
اللهم‌ اغفِرْ ‌لي‌ ذنوبي ‌و ‌خطايايَ ‌كلَّها، ‌اللهم ‌أَنعِشْني ‌و اجبُرْني، ‌و اهدِني‌ لصالحِ ‌الأعمالِ و ‌الأخلاقِ؛ ‌فإنه ‌لا ‌يهدي‌ لصالحها ‌و‌ لا‌ يصرفُ ‌سيِّئَها‌ إلا‌ أنت;;
اللهُمَّ ‌‌إِنِّي ‌‌أَسْأَلُكَ ‌‌مِنْ ‌‌فَضْلِكَ ‌‌وَرَحْمَتِكَ ‌‌فَإِنَّهُ ‌‌لَا ‌‌يَمْلِكُهَا ‌‌إِلَّا أَنْتَ;;
اللَّهُمَّ‌‌ إنِّي‌‌ أَعُوذُ ‌‌بِكَ ‌‌مِنْ ‌‌شرِّ سَمْعِي، وَمِنْ ‌‌شرِّ بَصرِي، وَمِنْ ‌‌شرِّ لِسَانِي، وَمِنْ‌‌ شرِّ قَلْبِي ‌‌وَمِنْ ‌‌شرِّ ‌‌مَنِيِّي;;
اللَّهُمَّ ‌‌إنِّي ‌‌أَعُوذُ بِكَ ‌‌مِنَ ‌‌الجُوعِ ‌‌فَإنَّهُ ‌‌بِئْسَ ‌‌الضَّجِيعُ، وَأَعُوذُ بِكَ ‌‌مِنَ ‌‌الخِيَانَةِ ‌‌فَإنَّهَا ‌‌بِئْستِ ‌‌البِطَانَةُ;;
اللَّهُمَّ ‌‌إنِّي ‌‌أَسْأَلُكَ ‌‌يَا الله ‌‌بِأَنَّكَ ‌‌الوَاحِدُ الأَحَدُ الصَّمدُ، الَّذِي ‌‌لَمْ‌‌ يَلِدْ ‌‌وَلَمْ‌‌ يُولَدْ، وَلَمْ ‌‌يَكُنْ ‌‌لَهُ ‌‌كُفُواً أَحَدٌ أَنْ ‌‌تَغْفِرَ لِي‌‌ ذُنُوبِي ‌‌إنَّكَ ‌‌أَنْتَ‌‌ الغَفُورُ الرَّحِيمُ;;
اللّهُمّ بَلّغْنَا رَمَضَان وَارْزُقْنَا حُسْنِ الْعَمَلِ فِيهِ وَتَقْبلَه مِنَّا يَا رَبِّ الْعَالَمِينَ;;
اللهمَّ بَلِّغْنَا رَمَضَانَ لا فَاقِدِينَ وَلَا مَفْقُودِينَ;;
اللّهُمّ بَلِّغْنَا رَمَضَان وأعنا عَلَى الصِّيَامِ وَالصَّلَاةِ وَالْقِيَام وَقِرَاءَةِ الْقُرْآنِ;;
اللّهُمّ بَلِّغْنَا رَمَضَان ، بَلاغٌ قَبُول وَتَوْفِيق وَنَحْن وَجَمِيع أَهْلِنَا وأحبتنا فِي أَحْسنِ حال;;
أَنِّي‌ ‌‌مَسَّنِيَ ‌‌الضُّرُّ وَأَنتَ ‌‌أَرْحَمُ ‌‌الرَّاحِمِينَ;;
وَمَن‌‌ يَتَوَكَّلْ ‌‌عَلَى‌‌ اللَّهِ ‌‌فَهُوَ ‌‌حَسْبُهُ‌ ۚ;;
وَتَوَكَّلْ‌‌ ‌عَلَى‌ ‌اللَّهِ ‌‌وَكَفَى‌ ‌بِاللَّهِ ‌‌وَكِيلًا;;
الْحَمْدُ‌ لِلَّهِ ‌‌حَمْدًا ‌ڪَثِيرًا ‌طَيِّبًا ‌‌مُبَارَڪََا‌ ‌فِيهِ;;
الحَمْدُ ‌لِلّهْ ‌عَلَى ‌كُلِّ‌ حَالْ;;
رَّبِّ‌ اغْفِرْ وَارْحَمْ ‌‌وَأَنتَ ‌‌خَيْرُ الرَّاحِمِينَ;;
إِنَّ ‌‌مَعَ ‌‌ٱلْعُسْرِ ‌‌يُسْراً;;
اللَّهُ‌‌ ‌لَا‌‌ إ‌ِلَهَ‌‌ ‌إِلَّا‌‌ ‌هُوَ | اللَّهُ‌‌ ‌لَا‌‌ إ‌ِلَهَ‌‌ ‌إِلَّا‌‌ ‌هُوَ‌‌ ‌الْحيُّ ‌‌الْقَيُّوم;;
سُبْحَانَ ‌‌الْلَّهِ، وَالْحَمْدُ ‌لِلَّهِ، وَلَا ‌‌إِلَهَ‌ ‌‌إ‌ِلَّا ‌‌الْلَّهُ، وَالْلَّهُ ‌‌أَكْبَرُ;;
أَسْتَغْفِرُ اللَّهَ ‌‌وَأَتُوبُ ‌‌إِلَيْهِ;;
اللّٰهُ‌ ‌أَكْبر;;
لَاْ ‌إِلٰهَ ‌إِلَّاْ‌ اللّٰه;;
أَسْتَغْفِرُ‌ اللّٰه;;
الْحَمْدُ‌ لِلّٰه;;
سُبْحَانَ اللّٰه;;
إِنْ‌ شَاءَ ‌اللّٰه;;
مَاشَاءَ‌ اللّٰه;;
بَاْرَكَ‌‌ اللّٰهُ‌ ‌‌فِیكْ;;
جَزَاكَ‌‌ اللّٰهُ ‌‌خَيْرََا;;
وَ‌ فِیكُمْ ‌بَاْرَكَ ‌اللّٰه;;
وَ ‌جَزَاكُمْ‌‌ اللّٰهُ ‌‌خَيْرََا;;
وَ ‌إِيَّاكُمْ;;
إِنَّ‌‌ ‌رَبِّى‌‌ لَسمِيعُ | إِنَّ‌‌ ‌رَبِّى‌‌ لَسمِيعُ‌ ‌‌ٱلدُّعَآءِ;;

[quran_verse]
رَبَّنَا ‌اغْفِرْ‌ ‌لَنَا ‌وَلِإِخْوَانِنَا ‌‌الَّذِينَ ‌سَبَقُونَا ‌بِالْإِيمَانِ ‌وَلَا ‌تَجْعَلْ ‌فِي ‌قُلُوبِنَا ‌غِلًّا ‌لِلَّذِينَ ‌آمَنُوا ‌رَبَّنَا ‌إِنَّكَ ‌رَءُوفٌ ‌رَحِيمٌ;;
رَبَّنَا ‌آتِنَا‌ فِي ‌الدُّنْيَا ‌حَسَنَةً ‌وَفِي ‌الْآخِرَةِ ‌حَسَنَةً ‌وَقِنَا‌ عَذَابَ ‌النَّارِ;;
رَبَّنَا ‌ظَلَمْنَا‌ أَنْفُسَنَا ‌وَإِنْ ‌لَمْ ‌‌تَغْفِرْ‌ لَنَا ‌وَتَرْحَمْنَا ‌لَنَكُونَنَّ‌ مِنَ ‌الْخَاسِرِينَ;;
رَبِّ‌‌ ‌اغْفِرْ‌ ‌لي‌ ‌‌خَطِيئَتي‌‌ وجَهْلِي‌، ‌وإسْرَافِي‌‌ ‌في‌‌ ‌أمْرِي‌‌ ‌كُلِّهِ‌، ‌وما‌‌ ‌أنْتَ‌‌ ‌أعْلَمُ‌‌ ‌به‌‌ ‌مِنِّي‌، ‌اللَّهُمَّ‌ ‌‌اغْفِرْ‌‌ ‌لي‌ ‌‌خَطَايَايَ‌، ‌وعَمْدِي‌‌ وجَهْلِي‌ ‌وهَزْلِي‌، ‌وكُلُّ ‌‌ذلكَ ‌‌عِندِي‌، ‌اللَّهُمَّ‌ ‌اغْفِرْ‌‌ لي‌ ‌ما ‌‌قَدَّمْتُ ‌‌وما‌ ‌أخَّرْتُ‌، ‌وما‌ ‌أسْرَرْتُ ‌‌وما ‌‌أعْلَنْتُ‌، ‌أنْتَ‌ ‌‌المُقَدِّمُ‌ ‌وأَنْتَ‌ ‌المُؤَخِّرُ‌، ‌‌وأَنْتَ‌‌ علَى‌ ‌كُلِّ ‌‌شيءٍ ‌‌قَدِيرٌ;;
قُلْ‌ ‌أَبِاللَّهِ‌ ‌‌وَآيَاتِهِ‌ ‌‌وَرَسُولِهِ‌ ‌‌كُنْتُمْ‌ ‌‌تَسْتَهْزِئُونَ;;
رَبَّنَا‌‌ أَفْرِغْ ‌‌عَلَيْنَا‌ ‌‌صَبْرًا وَثَبِّتْ ‌‌أَقْدَامَنَا ‌‌وَانصُرْنَا ‌‌عَلَى ‌‌الْقَوْمِ ‌‌الْكَافِرِينَ;;
رَبَّنَا ‌‌إِنَّنَا آمَنَّا ‌‌فَاغْفِرْ لَنَا ‌‌ذُنُوبَنَا ‌‌وَقِنَا ‌‌عَذَابَ ‌‌النَّارِ;;
رَبَّنَا ‌‌آتِنَا ‌‌مِن ‌‌لَّدُنكَ ‌‌رَحْمَةً ‌‌وَهَيِّئْ ‌‌لَنَا ‌‌مِنْ ‌‌أَمْرِنَا ‌‌رَشَدًا;;
رَبَّنَا اغْفِرْ لِي ‌‌وَلِوَالِدَيَّ ‌‌وَلِلْمُؤْمِنِينَ ‌‌يَوْمَ ‌‌يَقُومُ ‌‌الْحِسَابُ;;
ربَّنَا‌ اغْفِرْ‌ لَنَا‌ ذُنُوبَنَا‌ وَإِسْرَافَنَا ‌فِي ‌أَمْرِنَا‌ وَثَبِّتْ ‌أَقْدَامَنَا ‌وانصُرْنَا‌ عَلَى‌ الْقَوْمِ‌ الْكَافِرِينَ;;
رَبَّنَا‌ لَا‌ تَجْعَلْنَا‌ ‌‌فِتْنَةً‌‌ لِلْقَوْمِ ‌‌الظَّالِمِينَ ‌ وَنَجِّنَا ‌بِرَحْمَتِكَ‌ مِنَ ‌الْقَوْمِ‌ الْكَافِرِينَ;;
رَبَّنَا ا‌صْرِفْ ‌عَنَّا ‌عَذَابَ ‌جَهَنَّمَ ‌إِنَّ ‌عَذَابَهَا ‌كَانَ ‌غَرَامًا  ‌إِنَّهَا‌ سَاءَتْ ‌مُسْتَقَرًّا ‌وَمُقَامًا;;
رَبَّنَا ‌عَلَيْكَ ‌تَوَكَّلْنَا ‌وَإِلَيْكَ ‌أَنَبْنَا ‌وَإِلَيْكَ ‌الْمَصِيرُ;;
رَبَّنَا‌ مَا ‌خَلَقْتَ ‌هذَا ‌بَاطِلًا ‌سُبْحَانَكَ ‌فَقِنَا ‌عَذَابَ ‌النَّارِ;;
رَبَّنَا ‌لَا تُؤَاخِذْنَا ‌إِنْ ‌نَسِينَا ‌أَوْ أَخْطَأْنَا ‌رَبَّنَا ‌وَلَا‌ تَحْمِلْ‌ عَلَيْنَا ‌إِصْرًا‌ كَمَا ‌حَمَلْتَهُ ‌عَلَى ‌الَّذِينَ ‌مِنْ ‌قَبْلِنَا ‌رَبَّنَا ‌وَلَا‌ تُحَمِّلْنَا‌ مَا ‌لَا ‌طَاقَةَ‌ ‌لَنَا‌ بِهِ ‌وَاعْفُ ‌عَنَّا ‌وَاغْفِرْ ‌لَنَا ‌وَارْحَمْنَا ‌أَنْتَ ‌مَوْلَانَا ‌فَانْصُرْنَا ‌عَلَى‌ الْقَوْمِ‌ الْكَافِرِينَ;;
رَبَّنَا‌ لَا‌ تُزِغْ ‌قُلُوبَنَا ‌بَعْدَ ‌إِذْ ‌هَدَيْتَنَا ‌وَهَبْ ‌لَنَا ‌مِنْ ‌لَدُنْكَ ‌رَحْمَةً ‌إِنَّكَ‌ أَنْتَ‌ الْوَهَّابُ‌ * ‌رَبَّنَا‌ إِنَّكَ ‌جَامِعُ ‌النَّاسِ ‌لِيَوْمٍ ‌لَا‌ رَيْبَ ‌فِيهِ ‌إِنَّ ‌اللَّهَ ‌لَا يُخْلِفُ ‌الْمِيعَاد‌;;
رَبَّنَا ‌هَبْ ‌لَنَا ‌مِنْ ‌أَزْوَاجِنَا ‌وَذُرِّيَّاتِنَا ‌قُرَّةَ ‌أَعْيُنٍ ‌وَاجْعَلْنَا‌ لِلْمُتَّقِينَ‌ إِمَامًا;;
رَبَّنَا ‌آمَنَّا ‌‌بِمَا ‌أَنْزَلْتَ ‌‌وَاتَّبَعْنَا ‌‌الرَّسُولَ ‌‌فَاكْتُبْنَا ‌‌مَعَ ‌‌الشَّاهِدِينَ;;
رَبَّنَا أَفْرِغْ ‌‌عَلَيْنَا ‌‌صَبْرًا وَتَوَفَّنَا ‌‌مُسْلِمِينَ;;
رَبَّنَا أَتْمِمْ ‌‌لَنَا نُورَنَا ‌‌وَاغْفِرْ لَنَا ‌‌إِنَّكَ ‌‌عَلَى ‌‌كُلِّ ‌‌شَيْءٍ ‌‌قَدِيرٌ;;
رَبَّنَا‌‌ آمَنَّا ‌‌فَاغْفِرْ لَنَا ‌‌وَارْحَمْنَا‌‌ وَأَنتَ ‌‌خَيْرُ ‌‌الرَّاحِمِينَ;;
رَبَّنَا اغْفِرْ لَنَا ‌‌ذُنُوبَنَا ‌‌وَإِسْرَافَنَا ‌‌فِي ‌‌أَمْرِنَا ‌‌وَثَبِّتْ ‌‌أَقْدَامَنَا ‌‌وَانصُرْنَا ‌‌عَلَى ‌‌الْقَوْمِ ‌‌الْكَافِرِينَ;;
رَبَّنَا ‌‌لَا تَجْعَلْنَا ‌‌فِتْنَةً ‌‌لِلَّذِينَ ‌‌كَفَرُوا ‌وَاغْفِرْ‌ لَنَا‌‌ ‌رَبَّنَا‌ إِنَّكَ ‌‌أَنْتَ ‌‌الْعَزِيزُ‌ الْحَكِيمُ;;
رَبِّ اجْعَلْنِي‌‌ مُقِيمَ ‌‌الصَّلَاةِ ‌‌وَمِنْ ‌‌ذُرِّيَّتِي ‌‌رَبَّنَا ‌‌وَتَقَبَّلْ ‌‌دُعَاءِ;;
رَبِّ ‌‌ابْنِ ‌‌لِي ‌‌عِنْدَكَ ‌‌بَيْتًا ‌‌فِي ‌‌الْجَنَّةِ;;
رَبِّ ‌‌إِنِّي ‌‌لِمَا أَنزَلْتَ ‌‌إِلَيَّ ‌‌مِنْ ‌‌خَيْرٍ ‌‌فَقِيرٌ;;
رَبِّ‌‌ إِنِّي‌‌ ظَلَمْتُ‌‌ نَفْسِي‌‌ فَاغْفِرْ لِي;;
رَبِّ‌ ‌‌لَا‌ ‌تَذَرْنِي ‌‌فَرْدًا ‌‌وَأَنْتَ ‌‌خَيْرُ ‌الْوَارِثِينَ;;
رَبِّ ‌‌هَبْ ‌‌لِي ‌‌مِن ‌‌لَّدُنكَ ‌‌ذُرِّيَّةً ‌‌طَيِّبَةً ‌‌إِنَّكَ ‌‌سَمِيعُ ‌‌الدُّعَاءِ;;
رَبِّ‌‌ إِنِّي‌‌ أَعُوذُ بِكَ ‌‌أَنْ ‌‌أَسْأَلَكَ‌‌ مَا لَيْسَ‌‌ لِي‌‌ بِهِ ‌‌عِلْمٌ ‌‌وَإِلَّا ‌‌تَغْفِرْ لِي ‌‌وَتَرْحَمْنِي‌‌ أَكُن‌‌ مِّنَ ‌‌الْخَاسِرِينَ;;
رَبِّ ‌‌هَبْ ‌‌لِي‌‌ ‌حُكْمًا‌‌ ‌وَأَلْحِقْنِي‌‌ ‌بِالصَّالِحِينَ‌ * ‌وَاجْعَلْ ‌‌لِي‌‌ ‌لِسَانَ‌‌ ‌صِدْقٍ ‌‌فِي‌‌ ‌الْآخِرِينَ‌ * ‌وَاجْعَلْنِي‌ ‌‌مِنْ ‌‌وَرَثَةِ ‌‌جَنَّةِ ‌‌النَّعِيمِ;;
رَبِّ‌ أَعُوذُ‌ بِكَ ‌‌مِنْ ‌‌هَمَزَاتِ‌ ‌‌الشَّيَاطِينِ ‌‌‌وَأَعُوذُ‌ بِكَ ‌‌رَبِّ‌ ‌‌أَنْ ‌‌يَحْضُرُونِ;;
رَبِّ‌ ‌‌أَوْزِعْنِي‌ ‌‌أَنْ‌ ‌‌أَشْكُرَ ‌نِعْمَتَكَ‌ ‌‌الَّتِي‌‌ أَنْعَمْتَ‌ ‌‌عَلَيَّ‌ ‌‌وَعَلَى ‌‌وَالِدَيَّ ‌‌وَأَنْ ‌‌أَعْمَلَ ‌‌صَالِحًا ‌‌تَرْضَاهُ ‌‌وَأَدْخِلْنِي ‌‌بِرَحْمَتِكَ‌ ‌‌فِي ‌‌عِبَادِكَ ‌‌الصَّالِحِينَ;;
رَبِّ ‌إِنِّي ‌‌أَعُوذُ ‌بِكَ ‌‌أَنْ‌ ‌‌أَسْأَلَكَ ‌‌مَا لَيْسَ‌‌ ‌لِي‌‌ ‌بِهِ ‌‌عِلْمٌ‌‌ ‌وَإِلَّا‌ تَغْفِرْ ‌لِي‌‌ ‌وَتَرْحَمْنِي‌‌ ‌أكُنْ‌‌ ‌مِنَ ‌‌الْخَاسِرِينَ;;
رَبِّ‌ اشْرَحْ‌‌ لِي‌ صَدْرِي‌ * ‌وَيَسِّرْ ‌لِي ‌أَمْرِي‌ * ‌وَاحْلُلْ ‌عُقْدَةً ‌مِنْ‌ لِسَانِي‌ * ‌يَفْقَهُوا‌ قَوْلِي;;
رَبِّ ‌أَدْخِلْنِي‌‌ ‌مُدْخَلَ‌ ‌‌صِدْقٍ ‌‌وَأَخْرِجْنِي‌‌ مُخْرَجَ ‌‌صِدْقٍ ‌‌وَاجْعَلْ ‌‌لِي‌‌ ‌مِنْ‌ ‌‌لَدُنْكَ ‌‌سُلْطَانًا‌ ‌‌نَصِيرًا;;
رَبِّ ‌اجْعَلْنِي‌‌ ‌مُقِيمَ‌‌ ‌الصَّلَاةِ ‌‌‌وَمِنْ‌ ‌‌ذُرِّيَّتِي‌‌ رَبَّنَا ‌وَتَقَبَّلْ ‌‌دُعَاء;;
رَبِّ‌ ‌‌بِمَا‌ أَنْعَمْتَ‌‌ ‌عَلَيَّ‌‌ فَلَنْ‌‌ ‌أَكُونَ ‌‌ظَهِيرًا ‌‌لِلْمُجْرِمِينَ;;
رَّبِّ ‌‌زِدْنِي‌‌ عِلْمًا;;
إِنَّا‌‌ لِلّهِ‌‌ ‌وَإِنَّـا‌‌ إِلَيْهِ‌‌ ‌رَاجِعونَ‌‌ اَللَّهُمَّ‌ أَجِرْنِي‌ فِي‌ مُصِيبَتِي، ‌وَأَخْلَفَ ‌لِي ‌خَيْرًا‌ مِنْهَا;;
فَتَوَكَّلْ‌ ‌‌عَلَى‌‌ ٱللَّهِ‌ ‌‌إِنَّكَ ‌‌عَلَى‌‌ ٱلْحَقِّ‌ ‌‌ٱلْمُبِينِ;;
وَلَقَدْ‌ يَسَّرْنَا‌ الْقُرْآنَ‌ لِلذِّكْرِ‌ فَهَلْ ‌مِنْ ‌مُدَّكِر;;
قُل‌ لَّن ‌يُصِيبَنَآ ‌إِلَّا ‌مَا ‌كَتَبَ ‌ٱللَّهُ ‌لَنَا‌ هُوَ‌ مَوْلَـٰنَا;;
وَالذَّاكِرِينَ‌ اللَّهَ ‌كَثِيراً‌ وَالذَّاكِرَاتِ‌ أَعَدَّ ‌اللَّهُ ‌لَهُمْ ‌مَغْفِرَةً ‌وَأَجْراً ‌عَظِيماً;;
مَآ أَصَابَ ‌مِن ‌مُّصِيبَةٍ‌ إِلَّا ‌بِإِذۡنِ‌ ٱللَّهِۗ‌ وَمَن ‌يُؤۡمِنۢ‌ بِٱللَّهِ‌ يَهۡدِ‌ قَلۡبَهُۥۚ‌ وَٱللَّهُ‌ بِكُلِّ‌ شَيۡءٍ ‌عَلِيمٞ;;
وَإِذَا‌ سَأَلَكَ‌ عِبَادِي ‌عَنِّي ‌فَإِنِّي ‌قَرِيبٌ;;
أَنَّ‌ ٱلْقُوَّةَ‌ لِلَّهِ ‌جَمِيعًا;;
فَمَنْ‌ ‌‌يَكْفُرْ‌ بِالطَّاغُوتِ‌ ‌‌وَيُؤْمِنْ‌‌ بِاللَّهِ‌ ‌‌فَقَدِ‌ اسْتَمْسَكَ‌‌ بِالْعُرْوَةِ‌ الْوُثْقَى ‌‌لا‌‌ ‌انفِصَامَ‌‌ لَهَا;;
وَ‌ رَحْمَتِى ‌‌وَسِعَتْ ‌‌كُلَّ ‌‌شَىْءٍ;;
وَ‌ أَتِمُّوا۟‌ ٱلْحَجَّ ‌‌وَٱلْعُمْرَةَ‌‌ لِلَّهِ;;
وَٱلْآخِرَةُ ‌‌خَيْرٌ‌ ‌‌وَأَبْقَىٰ‌ٓ;;
